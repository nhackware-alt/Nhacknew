-- ============================================================
-- PROFIT METER
-- ============================================================
local Profit = {
Sample = {}, -- timestamps of successful cuts
WindowSize = 30, -- seconds window for rate calc
}

local function GetPing()
local ok, value = pcall(function()
return Stats.Network.ServerStatsItem["Data Ping"]:GetValue() / 1000
end)
if ok and value and value > 0 then return value end
return CONFIG.PING_FALLBACK
end

local function RecordCut()
table.insert(Profit.Sample, tick())
-- prune old
local cutoff = tick() - Profit.WindowSize
while #Profit.Sample > 0 and Profit.Sample[1] < cutoff do
table.remove(Profit.Sample, 1)
end
end

local function GetRates()
local n = #Profit.Sample
local ping = GetPing()
local cycle = CONFIG.STEP_DELAY + CONFIG.POST_CUT_WAIT + CONFIG.MAIN_LOOP_DELAY + ping

-- rate from samples
local ratePerSec, ratePerMin, perHour
if n >= 2 then
local span = Profit.Sample[n] - Profit.Sample[1]
if span > 0 then
ratePerSec = (n - 1) / span
else
ratePerSec = 0
end
else
-- nominal from cycle
ratePerSec = cycle > 0 and (1 / cycle) or 0
end

ratePerMin = ratePerSec * 60
perHour = ratePerSec * 3600 * CONFIG.RICE_PRICE

return {
cycle = cycle,
ping = ping,
ratePerSec = ratePerSec,
ratePerMin = ratePerMin,
perHour = perHour,
samples = n,
}
end

local function GetTool(name)
local ch = Player.Character
if not ch then return nil end
local hum = ch:FindFirstChildOfClass("Humanoid")
if not hum then return nil end
local tool = ch:FindFirstChild(name) or Player.Backpack:FindFirstChild(name)

end

-- ============================================================
-- SEAT HELPERS
-- ============================================================
local function IsSeatedAt(hum, seat)
return seat and seat.Occupant == hum
end

local function SitAndWait(hum, seat, timeout)
local ch = hum.Parent
local hrp = ch and ch:FindFirstChild("HumanoidRootPart")
if not hrp then return false end

if hum.Sit then
hum.Sit = false
local t0 = tick()
while hum.Sit and tick() - t0 < 0.5 do task.wait(0.05) end
end

hrp.CFrame = seat.CFrame * CFrame.new(0, 3, 0)
task.wait(0.1)

pcall(function() seat:Sit(hum) end)

local t0 = tick()
while tick() - t0 < timeout do
if IsSeatedAt(hum, seat) then return true end
task.wait(0.05)
end
return false
end

local function WaitForTool(names, ch, timeout)
local t0 = tick()
while tick() - t0 < timeout do
for _, name in ipairs(names) do
local t = Player.Backpack:FindFirstChild(name) or ch:FindFirstChild(name)
if t and t:IsA("Tool") then return t end
end
task.wait(0.1)
end
return nil
end

-- ============================================================
-- AUTO EAT - RICE
-- ============================================================
local function AutoEatRice(forceBasic)
local ch = Player.Character
if not ch then return false end
local hum = ch:FindFirstChildOfClass("Humanoid")
if not hum or hum.Health <= 0 then return false end

local names = forceBasic and {"Basic Food"} or {"Best Food", "Good Food", "Basic Food"}
local tool = nil
for _, name in ipairs(names) do
local t = Player.Backpack:FindFirstChild(name) or ch:FindFirstChild(name)
if t and t:IsA("Tool") then tool = t break end
end

if not tool then
local Stores = Workspace:FindFirstChild("Stores")
local Food = Stores and Stores:FindFirstChild("Food")
local Seats = Food and Food:FindFirstChild("CustomerSeats")
if not Seats then return false end

local priority = forceBasic and {"1"} or {"1", "2", "3"}
local seated = false
for _, seatName in ipairs(priority) do
local seat = Seats:FindFirstChild(seatName)
if seat and seat:IsA("Seat") then
if SitAndWait(hum, seat, CONFIG.EAT_SEAT_TIMEOUT) then
seated = true
break
end
end
end
if not seated then return false end

tool = WaitForTool(names, ch, CONFIG.EAT_TOOL_TIMEOUT)
if not tool then return false end
end

pcall(function() hum:EquipTool(tool) end)
task.wait(0.05)
local ok = pcall(function() tool:Activate() end)
task.wait(0.4)

-- free the seat
local seatPart = hum.SeatPart
if seatPart and seatPart:IsA("Seat") then
pcall(function() seatPart:Sit(nil) end)
end

return ok
end

-- ============================================================
-- GET RICE
-- ============================================================
local function GetRice()
if not State.RiceFolder then
State.RiceFolder = Workspace:FindFirstChild("Rice")
end
if not State.RiceFolder then return nil end

local Karts = Workspace:FindFirstChild("Karts")
local kart = Karts and Karts:FindFirstChild(PlayerName)
local kartPos = kart and kart.PrimaryPart and kart.PrimaryPart.Position
if not kartPos then return nil end

local live = {}
for _, rice in ipairs(State.RiceFolder:GetChildren()) do
if rice:IsA("Model") then
local hp = rice:GetAttribute("Health")
local rw = rice:GetAttribute("Reward")
local pp = rice.PrimaryPart
if hp and hp > 0 and rw and rw > 0 and pp then
table.insert(live, {
rice = rice,
pp = pp,
dist = (pp.Position - kartPos).Magnitude,
})
end
end
end

if #live == 0 then return nil end
table.sort(live, function(a, b) return a.dist < b.dist end)

for _, entry in ipairs(live) do
if entry.rice ~= State.LastRice then
State.LastRice = entry.rice
return entry.rice, entry.pp
end
end

State.LastRice = live[1].rice
return live[1].rice, live[1].pp
end

-- ============================================================
-- USE KART
-- ============================================================
local function UseKart(kart)
if kart.Name ~= PlayerName then return end

local seat = kart:FindFirstChildOfClass("VehicleSeat")
if not seat then return end
local kartRoot = kart.PrimaryPart
if not kartRoot then return end

while kart.Parent and State.Enabled and not State.Paused do
local ch = Player.Character
local hum = ch and ch:FindFirstChildOfClass("Humanoid")
if not hum or hum.Health <= 0 then break end

if seat.Occupant ~= hum then
pcall(function() seat:Sit(hum) end)
end

local rice, part = GetRice()
if rice and part then
local hpBefore = rice:GetAttribute("Health") or 0

pcall(function()
kartRoot.AssemblyLinearVelocity = Vector3.zero
kartRoot.AssemblyAngularVelocity = Vector3.zero
kart:PivotTo(part.CFrame * CFrame.new(CONFIG.OFFSET_X, CONFIG.OFFSET_Y, CONFIG.OFFSET_Z))
kartRoot.AssemblyLinearVelocity = Vector3.zero
kartRoot.AssemblyAngularVelocity = Vector3.zero
end)

task.wait(CONFIG.STEP_DELAY)

local hpAfter = rice:GetAttribute("Health") or 0
if hpAfter < hpBefore then
State.RiceCount = State.RiceCount + 1
RecordCut()
else
task.wait(CONFIG.POST_CUT_WAIT)
pcall(function()
kart:PivotTo(part.CFrame * CFrame.new(CONFIG.OFFSET_X, CONFIG.OFFSET_Y, CONFIG.OFFSET_Z))
end)
task.wait(0.1)
end

State.LastRice = rice
else
task.wait(CONFIG.NO_RICE_WAIT)
end

task.wait(CONFIG.MAIN_LOOP_DELAY)
end
end

-- ============================================================
-- RAGDOLL SAVE
-- ============================================================
local function GetKart()
local Karts = Workspace:FindFirstChild("Karts")
return Karts and Karts:FindFirstChild(PlayerName)
end

local function IsOnFoodSeat()
local ch = Player.Character
local hum = ch and ch:FindFirstChildOfClass("Humanoid")
if not hum then return false end
local seat = hum.SeatPart
if not seat then return false end
return not seat:IsA("VehicleSeat")
end

local function LiftKart()
local kart = GetKart()
if not kart or not kart.PrimaryPart then return end

local root = kart.PrimaryPart
RagdollState.OriginalCFrame = root.CFrame

if RagdollState.BodyPosition then
RagdollState.BodyPosition:Destroy()
end

local bp = Instance.new("BodyPosition")
bp.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
bp.P = 50000
bp.D = 1000
bp.Position = root.Position + Vector3.new(0, CONFIG.RAGDOLL_LIFT, 0)
bp.Parent = root
RagdollState.BodyPosition = bp

-- try to keep player seated
if RagdollState.MonitorThread then
task.cancel(RagdollState.MonitorThread)
RagdollState.MonitorThread = nil
end

RagdollState.MonitorThread = task.spawn(function()
while RagdollState.Count > 0 and State.Enabled do
local k = GetKart()
if not k or not k.Parent then break end
local s = k:FindFirstChildOfClass("VehicleSeat")
local ch = Player.Character
local hum = ch and ch:FindFirstChildOfClass("Humanoid")
if s and hum and hum.Health > 0 and s.Occupant ~= hum then
local hrp = ch:FindFirstChild("HumanoidRootPart")
if hrp then hrp.CFrame = s.CFrame * CFrame.new(0, 3, 0) end
pcall(function() s:Sit(hum) end)
end
task.wait(CONFIG.RAGDOLL_RETRY)
end
RagdollState.MonitorThread = nil
end)
end

local function DropKart()
if RagdollState.BodyPosition then
RagdollState.BodyPosition:Destroy()
RagdollState.BodyPosition = nil
end
if RagdollState.MonitorThread then
task.cancel(RagdollState.MonitorThread)
RagdollState.MonitorThread = nil
end
end

local function HookRagdoll()
local Remotes = ReplicatedStorage:WaitForChild("Remotes", 10)
local ev = Remotes and Remotes:WaitForChild("RagdollEvent", 10)
if not ev then return end

pcall(function()
ev.OnClientEvent:Connect(function(rstate)
if not CONFIG.RAGDOLL_SAVE or not State.Enabled then return end

-- intentional exit (eating) — ignore
if rstate == true and State.IntentionalExit
and (tick() - State.IntentionalExitTime) < CONFIG.INTENTIONAL_EXIT_TIME then
return
end

-- on food seat — ignore
if rstate == true and IsOnFoodSeat() then return end

if rstate == true then
RagdollState.Count = RagdollState.Count + 1
if not RagdollState.Active then
RagdollState.Active = true
State.Paused = true
LiftKart()
end
else
RagdollState.Count = math.max(0, RagdollState.Count - 1)
if RagdollState.Count == 0 then
task.delay(CONFIG.RAGDOLL_RESTORE_DELAY, function()
if RagdollState.Count == 0 and RagdollState.Active then
RagdollState.Active = false
DropKart()
State.Paused = false
end
end)
end
end
end)
end)
end

-- ============================================================
-- AUTO EAT LOOP
-- ============================================================
local eatLoop = nil

local function StartEatLoop()
if eatLoop then return end
eatLoop = task.spawn(function()
while State.Enabled do
local stats = Player:FindFirstChild("stats")
local hunger = stats and stats:FindFirstChild("Hunger")
local ch = Player.Character
local hum = ch and ch:FindFirstChildOfClass("Humanoid")

if hunger and hum and hum.Health > 0 and hunger.Value < CONFIG.HUNGER_MIN then
State.Paused = true
State.IntentionalExit = true
State.IntentionalExitTime = tick()
State.EatStatus = "голоден, иду есть"

hum.Sit = false
local t0 = tick()
while hum.Sit and tick() - t0 < 1.5 do task.wait(0.05) end

local attempts = 0
local consecutiveFails = 0

while State.Enabled
and hunger.Value < CONFIG.HUNGER_MAX
and attempts < CONFIG.EAT_MAX_ATTEMPTS
and hum.Health > 0
do
attempts = attempts + 1
State.EatAttempts = attempts

local ok = AutoEat()

if ok then
consecutiveFails = 0
State.EatStatus = "Hunger=" .. math.floor(hunger.Value)
else
consecutiveFails = consecutiveFails + 1
State.EatStatus = "retry #" .. attempts .. " (fail " .. consecutiveFails .. ")"
end

local waitTime = CONFIG.EAT_WAIT_SHORT
if consecutiveFails > CONFIG.EAT_FAIL_THRESHOLD_LONG then
waitTime = CONFIG.EAT_WAIT_LONG
elseif consecutiveFails > CONFIG.EAT_FAIL_THRESHOLD_MED then
waitTime = CONFIG.EAT_WAIT_MED
end

task.wait(waitTime + math.random() * 0.2)
end

State.EatStatus = "наелся (Hunger=" .. math.floor(hunger.Value) .. ")"
State.IntentionalExit = false
State.Paused = false
task.wait(1)
end

task.wait(2)
end
eatLoop = nil
end)
end

local function StartFarm()
if State.Enabled then return end
State.Enabled = true
State.Error = nil
State.RiceCount = 0
State.LastRice = nil
State.EatStatus = ""
State.SessionStart = tick()
Profit.Sample = {}

end

local function StopFarm()
State.Enabled = false
State.Paused = false
RagdollState.Count = 0
RagdollState.Active = false
DropKart()
end

local function ToggleFarm()
if State.Enabled then StopFarm() else StartFarm() end
end

local function SaveConfig()
if not writefile then return false end
local ok, encoded = pcall(function() return game:GetService("HttpService"):JSONEncode(CONFIG) end)
if not ok then return false end
pcall(function() writefile(CFG_FILE, encoded) end)
return true
end

local function LoadConfig()
if not (isfile and readfile) then return false end
if not isfile(CFG_FILE) then return false end
local ok, content = pcall(function() return readfile(CFG_FILE) end)
if not ok then return false end
local ok2, decoded = pcall(function() return game:GetService("HttpService"):JSONDecode(content) end)
if not ok2 or type(decoded) ~= "table" then return false end
for k, v in pairs(decoded) do
if CONFIG[k] ~= nil and type(CONFIG[k]) == type(v) then
CONFIG[k] = v
end
end
return true
end

local function clamp(v)
if v < min then v = min end
if v > max then v = max end
return v
end

local function apply(v)
v = clamp(v)
if decimals then
local mult = 10 ^ decimals
v = math.round(v * mult) / mult
end
CONFIG[key] = v
box.Text = tostring(v)
end

minus.MouseButton1Click:Connect(function() apply(CONFIG[key] - step) end)
plus.MouseButton1Click:Connect(function() apply(CONFIG[key] + step) end)
box.FocusLost:Connect(function()
local n = tonumber(box.Text)
if n then apply(n) else box.Text = tostring(CONFIG[key]) end
end)

return { set = apply }
end
